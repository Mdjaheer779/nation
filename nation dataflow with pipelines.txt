CREATE OR REPLACE PROCEDURE NATION_EXPORT()
RETURNS VARCHAR
LANGUAGE SQL
AS
DECLARE

V_START_TIME TIMESTAMP;
V_END_TIME TIMESTAMP;

BEGIN

V_START_TIME := CURRENT_TIMESTAMP();

COPY INTO @AZURE_NATION_STAGE
FROM FACT_NATION
FILE_FORMAT = (
    TYPE = CSV
    FIELD_OPTIONALLY_ENCLOSED_BY = '"'
    FIELD_DELIMITER = '|'
)
HEADER = TRUE
SINGLE = TRUE
OVERWRITE = TRUE;

V_END_TIME := CURRENT_TIMESTAMP();

INSERT INTO NATION_EXPORT_AUDIT
(
    PROCESS_NAME,
    START_TIME,
    END_TIME,
    STATUS,
    MESSAGE
)
VALUES
(
    'NATION_EXPORT',
    :V_START_TIME,
    :V_END_TIME,
    'SUCCESS',
    'NATION DATA EXPORTED TO AZURE SUCCESSFULLY'
);

RETURN 'NATION DATA EXPORTED TO AZURE SUCCESSFULLY';

EXCEPTION
    WHEN OTHER THEN

        V_END_TIME := CURRENT_TIMESTAMP();

        INSERT INTO NATION_EXPORT_AUDIT
        (
            PROCESS_NAME,
            START_TIME,
            END_TIME,
            STATUS,
            MESSAGE
        )
        VALUES
        (
            'NATION_EXPORT',
            :V_START_TIME,
            :V_END_TIME,
            'FAILED',
            :SQLERRM
        );

        RETURN 'NATION DATA EXPORT FAILED: ' || SQLERRM;

END;
CALL NATION_EXPORT()
